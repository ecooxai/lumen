function(lumen_library output name)
    if(ANDROID)
        find_library(${output} NAMES ${name} PATHS "${LUMEN_PREFIX}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    else()
        find_library(${output} NAMES ${name} HINTS "${LUMEN_PREFIX}/lib" REQUIRED)
    endif()
    set(${output} "${${output}}" PARENT_SCOPE)
endfunction()

function(lumen_header output name)
    if(ANDROID)
        find_path(${output} NAMES ${name} PATHS "${LUMEN_PREFIX}/include" PATH_SUFFIXES "" v8 freetype2 harfbuzz webgpu NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    else()
        find_path(${output} NAMES ${name} HINTS "${LUMEN_PREFIX}/include" PATH_SUFFIXES v8 freetype2 harfbuzz REQUIRED)
    endif()
    set(${output} "${${output}}" PARENT_SCOPE)
endfunction()
