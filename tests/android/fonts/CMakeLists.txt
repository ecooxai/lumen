cmake_minimum_required(VERSION 3.22)
project(CodexAndroidFontTest LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set_property(GLOBAL PROPERTY JOB_POOLS codex_font_compile=2)
set(CMAKE_JOB_POOL_COMPILE codex_font_compile)
set(CMAKE_JOB_POOL_LINK codex_font_compile)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(PREFIX "${LUMEN_DEPS_DIR}/${ANDROID_ABI}")
if(NOT ANDROID OR NOT EXISTS "${PREFIX}/include/ft2build.h")
    message(FATAL_ERROR "Use the Android toolchain and an ABI-specific native dependency prefix")
endif()
add_library(codex_android_font_smoke SHARED
    font_test_jni.c font_preview.c ../font_smoke.c
    "${ROOT}/src/text/font.c" "${ROOT}/src/text/android_fonts.c"
    "${ROOT}/src/text/android_font_raster.c" "${ROOT}/src/base/util.c")
target_compile_definitions(codex_android_font_smoke PRIVATE LUMEN_FONT_SMOKE_JNI)
target_include_directories(codex_android_font_smoke PRIVATE "${PREFIX}/include" "${PREFIX}/include/harfbuzz")
set(FONT_LIBRARIES)
foreach(name freetype harfbuzz png16 bz2 brotlidec brotlicommon z)
    set(library "${PREFIX}/lib/lib${name}.a")
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing Android archive: ${library}")
    endif()
    list(APPEND FONT_LIBRARIES "${library}")
endforeach()
target_link_libraries(codex_android_font_smoke PRIVATE
    -Wl,--start-group ${FONT_LIBRARIES} -Wl,--end-group c++_shared android dl m)
target_link_options(codex_android_font_smoke PRIVATE -Wl,-z,max-page-size=16384)
